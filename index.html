How to Connect Google Sheets Sync

To have your phone and computer sync automatically via Google Drive, you can link this app to a Google Sheet using a free Google Apps Script web app:

  1. Go to sheets.new in your Google account to create a blank Google Sheet.
  2. Click on Extensions > Apps Script in the top menu.
  3. Delete any code in the editor and paste the Apps Script code provided below.
  4. Click Deploy > New deployment, select type Web app, set Execute as Me and Who has access to Anyone.
  5. Copy the generated Web App URL and paste it into the "Apps Script Web App URL" box in your app sidebar, then click Sync Now!
// Apps Script code snippet:
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  sheet.clear();
  sheet.appendRow(["JSON_DATA", JSON.stringify(data)]);
  return ContentService.createTextOutput(JSON.stringify({"status":"success"})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var val = sheet.getRange(1, 2).getValue();
  return ContentService.createTextOutput(val || "{}").setMimeType(ContentService.MimeType.JSON);
}
